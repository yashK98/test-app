import pandas as pd
from sqlalchemy import create_engine
from rapidfuzz.fuzz import ratio


# ============================================================
# 1. DATABASE CONNECTION
# ============================================================

DB_USER = "YOUR_USERNAME"
DB_PASSWORD = "YOUR_PASSWORD"
DB_HOST = "YOUR_HOST"
DB_PORT = "5432"
DB_NAME = "YOUR_DATABASE"

engine = create_engine(
    f"postgresql+psycopg2://{DB_USER}:{DB_PASSWORD}@{DB_HOST}:{DB_PORT}/{DB_NAME}"
)


# ============================================================
# 2. READ CANDIDATE PAIRS FROM POSTGRESQL
# ============================================================

query = """
SELECT *
FROM entity_recognition.contact_info_sp_features
"""

df = pd.read_sql(query, engine)

print("Number of candidate pairs:", len(df))
print("Columns:")
print(df.columns.tolist())


# ============================================================
# 3. STRING SIMILARITY FUNCTION
# ============================================================

def string_similarity(a, b):

    if pd.isna(a) or pd.isna(b):
        return 0.0

    a = str(a).strip()
    b = str(b).strip()

    if not a or not b:
        return 0.0

    return ratio(a, b) / 100.0


# ============================================================
# 4. EXACT MATCH FUNCTION
# ============================================================

def exact_match(a, b):

    if pd.isna(a) or pd.isna(b):
        return 0

    a = str(a).strip()
    b = str(b).strip()

    if not a or not b:
        return 0

    return int(a == b)


# ============================================================
# 5. CALCULATE FUZZY SIMILARITY FEATURES
# ============================================================

print("Calculating name similarity...")

df["name_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["name_1"],
        df["name_2"]
    )
]


print("Calculating native name similarity...")

df["native_name_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["native_name_1"],
        df["native_name_2"]
    )
]


print("Calculating address similarity...")

df["address_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["address_1"],
        df["address_2"]
    )
]


# ============================================================
# 6. CALCULATE EXACT MATCH FEATURES
# ============================================================

print("Calculating exact-match features...")


df["postcode_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["postcode_1"],
        df["postcode_2"]
    )
]


df["city_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["city_1"],
        df["city_2"]
    )
]


df["country_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["country_iso_1"],
        df["country_iso_2"]
    )
]


df["phone_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["phone_1"],
        df["phone_2"]
    )
]


df["website_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["website_1"],
        df["website_2"]
    )
]


df["email_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["email_1"],
        df["email_2"]
    )
]


# ============================================================
# 7. SELECT ML FEATURES
# ============================================================

feature_columns = [

    "id_1",
    "id_2",
