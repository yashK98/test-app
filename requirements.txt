import re
from bs4 import BeautifulSoup


def extract_email_content(html):

    if not html:
        return ""

    # --------------------------------------------------
    # Step 1: HTML -> BeautifulSoup
    # --------------------------------------------------

    soup = BeautifulSoup(html, "lxml")

    # Remove non-email-content HTML
    for tag in soup(["script", "style", "img", "svg", "head"]):
        tag.decompose()

    # --------------------------------------------------
    # Step 2: HTML -> plain text
    # --------------------------------------------------

    text = soup.get_text("\n")

    # --------------------------------------------------
    # Step 3: Remove quoted / previous emails
    # --------------------------------------------------

    reply_patterns = [
        r"(?im)^[-\s]*Original Message[-\s]*$",
        r"(?im)^Begin forwarded message:\s*$",
        r"(?im)^From:\s*.+$",
    ]

    positions = []

    for pattern in reply_patterns:
        match = re.search(pattern, text)

        if match:
            positions.append(match.start())

    if positions:
        text = text[:min(positions)]

    # --------------------------------------------------
    # Step 4: Remove signature
    # --------------------------------------------------

    signature_patterns = [
        r"(?im)^Regards,?\s*$",
        r"(?im)^Warm Regards,?\s*$",
        r"(?im)^Best Regards,?\s*$",
        r"(?im)^Kind Regards,?\s*$",
        r"(?im)^Thanks,?\s*$",
        r"(?im)^Thank you,?\s*$",
    ]

    positions = []

    for pattern in signature_patterns:
        match = re.search(pattern, text)

        if match:
            positions.append(match.start())

    if positions:
        text = text[:min(positions)]

    # --------------------------------------------------
    # Step 5: Remove known corporate noise
    # --------------------------------------------------

    noise_patterns = [
        r"(?im)^INTERNAL\s*$",
        r"(?im)^EXTERNAL EMAIL\s*$",
    ]

    for pattern in noise_patterns:
        text = re.sub(pattern, "", text)

    # --------------------------------------------------
    # Step 6: Clean whitespace
    # --------------------------------------------------

    lines = []

    for line in text.splitlines():

        line = line.strip()

        if line:
            lines.append(line)

    return "\n".join(lines)
