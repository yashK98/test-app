import pandas as pd
from rapidfuzz.fuzz import ratio


# ============================================================
# CONFIGURATION
# ============================================================

INPUT_FILE = "entity_resolution_training_data.csv"
OUTPUT_FILE = "entity_resolution_training_features.csv"


# ============================================================
# LOAD SYNTHETIC DATA
# ============================================================

df = pd.read_csv(INPUT_FILE)

print("Loaded rows:", len(df))


# ============================================================
# STRING SIMILARITY
# ============================================================

def string_similarity(a, b):

    if pd.isna(a) or pd.isna(b):
        return 0.0

    a = str(a).strip()
    b = str(b).strip()

    if not a or not b:
        return 0.0

    return ratio(a, b) / 100.0


# ============================================================
# EXACT MATCH
# ============================================================

def exact_match(a, b):

    if pd.isna(a) or pd.isna(b):
        return 0

    a = str(a).strip()
    b = str(b).strip()

    if not a or not b:
        return 0

    return int(a == b)


# ============================================================
# FUZZY FEATURES
# ============================================================

print("Calculating name similarity...")

df["name_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["name_1"],
        df["name_2"]
    )
]


print("Calculating native name similarity...")

df["native_name_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["native_name_1"],
        df["native_name_2"]
    )
]


print("Calculating address similarity...")

df["address_similarity"] = [
    string_similarity(a, b)
    for a, b in zip(
        df["address_1"],
        df["address_2"]
    )
]


# ============================================================
# EXACT-MATCH FEATURES
# ============================================================

print("Calculating exact-match features...")

df["postcode_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["postcode_1"],
        df["postcode_2"]
    )
]

df["city_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["city_1"],
        df["city_2"]
    )
]

df["country_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["country_iso_1"],
        df["country_iso_2"]
    )
]

df["phone_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["phone_1"],
        df["phone_2"]
    )
]

df["website_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["website_1"],
        df["website_2"]
    )
]

df["email_match"] = [
    exact_match(a, b)
    for a, b in zip(
        df["email_1"],
        df["email_2"]
    )
]


# ============================================================
# SELECT ML FEATURES
# ============================================================

feature_columns = [
    "name_similarity",
    "native_name_similarity",
    "address_similarity",
    "postcode_match",
    "city_match",
    "country_match",
    "phone_match",
    "website_match",
    "email_match"
]


ml_df = df[
    [
        "id_1",
        "id_2",
        *feature_columns,
        "label"
    ]
].copy()


# ============================================================
# CHECK DATA
# ============================================================

print("\n========================================")
print("FEATURE DATASET")
print("========================================")

print("Rows:", len(ml_df))

print("\nLabel distribution:")
print(ml_df["label"].value_counts())

print("\nSample:")
print(ml_df.head(20).to_string(index=False))


# ============================================================
# SAVE
# ============================================================

ml_df.to_csv(
    OUTPUT_FILE,
    index=False
)

print("\nSaved:", OUTPUT_FILE)
