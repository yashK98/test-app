import logging
from pathlib import Path

import pdfplumber
from docx import Document

logger = logging.getLogger(__name__)


class DocumentService:

    @staticmethod
    def extract_text(document_path: str) -> str:
        """
        Reads any supported document and extracts text.
        """

        extension = Path(document_path).suffix.lower()

        logger.info(f"Reading document : {document_path}")

        if extension == ".txt":
            return DocumentService._read_text(document_path)

        if extension == ".pdf":
            return DocumentService._read_pdf(document_path)

        if extension == ".docx":
            return DocumentService._read_docx(document_path)

        raise Exception(f"Unsupported document type : {extension}")

    @staticmethod
    def _read_text(path: str) -> str:

        with open(path, "r", encoding="utf-8") as file:
            return file.read()

    @staticmethod
    def _read_pdf(path: str) -> str:

        pages = []

        with pdfplumber.open(path) as pdf:

            for page in pdf.pages:

                text = page.extract_text()

                if text:
                    pages.append(text)

        return "\n".join(pages)

    @staticmethod
    def _read_docx(path: str) -> str:

        document = Document(path)

        paragraphs = []

        for paragraph in document.paragraphs:
            paragraphs.append(paragraph.text)

        return "\n".join(paragraphs)
