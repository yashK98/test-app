import re
from bs4 import BeautifulSoup


def extract_email_content(html):

    if not html:
        return ""

    # 1. HTML -> BeautifulSoup
    soup = BeautifulSoup(html, "lxml")

    # 2. Remove non-text content
    for tag in soup(["script", "style", "img", "svg", "head"]):
        tag.decompose()

    # 3. HTML -> text
    text = soup.get_text("\n")

    # 4. Remove external email warning
    text = re.sub(
        r"(?is)ATTENTION:\s*"
        r"This email came from an external source\..*?"
        r"our clients\.",
        "",
        text
    )

    # 5. Remove previous/quoted email
    reply_patterns = [
        r"(?im)^[-\s]*Original Message[-\s]*$",
        r"(?im)^Begin forwarded message:\s*$",
        r"(?im)^From:\s*.+$",
    ]

    positions = []

    for pattern in reply_patterns:
        match = re.search(pattern, text)

        if match:
            positions.append(match.start())

    if positions:
        text = text[:min(positions)]

    # 6. Remove signature
    signature_patterns = [
        r"(?im)^Regards,?\s*$",
        r"(?im)^Warm Regards,?\s*$",
        r"(?im)^Best Regards,?\s*$",
        r"(?im)^Kind Regards,?\s*$",
        r"(?im)^Thanks,?\s*$",
        r"(?im)^Thank you,?\s*$",
    ]

    positions = []

    for pattern in signature_patterns:
        match = re.search(pattern, text)

        if match:
            positions.append(match.start())

    if positions:
        text = text[:min(positions)]

    # 7. Remove INTERNAL banner
    text = re.sub(
        r"(?im)^INTERNAL\s*$",
        "",
        text
    )

    # 8. Clean whitespace
    lines = [
        line.strip()
        for line in text.splitlines()
        if line.strip()
    ]

    return "\n".join(lines)
