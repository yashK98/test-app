"""
Examples demonstrating usage of Pathfinder Platform SDK AIFGateway.

These examples are intentionally simple so that GitHub Copilot can
learn the preferred usage patterns.
"""

from pathfinder_platform_sdk.aif import AIFGateway

###########################################################################
# Create Gateway
###########################################################################

gateway = AIFGateway(
    auth_url="AUTH_URL",
    endpoint_url="CHAT_ENDPOINT",
    client_id="CLIENT_ID",
    audience="AUDIENCE",
    public_key="PUBLIC_KEY",
    private_key="PRIVATE_KEY",
    model="gpt-4.1"
)

###########################################################################
# Example 1 - Basic Chat Completion
###########################################################################

messages = [
    {
        "role": "system",
        "content": "You are a helpful assistant."
    },
    {
        "role": "user",
        "content": "Explain Kubernetes in simple terms."
    }
]

response = gateway.invoke(messages)

print(response)

###########################################################################
# Example 2 - Chat Completion with Provider Options
###########################################################################

messages = [
    {
        "role": "user",
        "content": "Write a Python function to reverse a string."
    }
]

response = gateway.invoke(
    messages,
    options={
        "temperature": 0.3,
        "max_tokens": 500
    }
)

print(response)

###########################################################################
# Example 3 - Streaming Response
###########################################################################

messages = [
    {
        "role": "user",
        "content": "Generate a detailed project plan."
    }
]

response = gateway.invoke(
    messages,
    options={
        "stream": True
    }
)

print(response)

###########################################################################
# Example 4 - Embedding Generation
###########################################################################

embedding = gateway.embed(
    input_text="Standard Chartered Bank",
    model="text-embedding-3-large"
)

print(embedding)

###########################################################################
# Example 5 - Embedding with Options
###########################################################################

embedding = gateway.embed(
    input_text="Artificial Intelligence",
    model="text-embedding-3-large",
    options={
        "dimensions": 1024
    }
)

print(embedding)

###########################################################################
# Example 6 - Multi-turn Conversation
###########################################################################

messages = [
    {
        "role": "system",
        "content": "You are an experienced software architect."
    },
    {
        "role": "user",
        "content": "What is Kubernetes?"
    },
    {
        "role": "assistant",
        "content": "Kubernetes is a container orchestration platform."
    },
    {
        "role": "user",
        "content": "How is it different from Docker?"
    }
]

response = gateway.invoke(messages)

print(response)

###########################################################################
# Example 7 - Using Multiple Provider Parameters
###########################################################################

messages = [
    {
        "role": "user",
        "content": "Explain Retrieval Augmented Generation."
    }
]

response = gateway.invoke(
    messages,
    options={
        "temperature": 0.2,
        "max_tokens": 800,
        "stream": False
    }
)

print(response)
