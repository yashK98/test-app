import pandas as pd
import joblib

from sklearn.model_selection import train_test_split
from sklearn.metrics import (
    classification_report,
    confusion_matrix,
    accuracy_score,
    precision_score,
    recall_score,
    f1_score
)

from xgboost import XGBClassifier


# ============================================================
# 1. LOAD DATA
# ============================================================

INPUT_FILE = "entity_resolution_training_features.csv"

df = pd.read_csv(INPUT_FILE)

print("Total records:", len(df))


# ============================================================
# 2. DEFINE FEATURES
# ============================================================

feature_columns = [
    "name_similarity",
    "native_name_similarity",
    "address_similarity",
    "postcode_match",
    "city_match",
    "country_match",
    "phone_match",
    "website_match",
    "email_match"
]

X = df[feature_columns]

y = df["label"]


# ============================================================
# 3. TRAIN / TEST SPLIT
# ============================================================

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)


print("\n========================================")
print("DATA SPLIT")
print("========================================")

print("Training records:", len(X_train))
print("Testing records:", len(X_test))

print("\nTraining labels:")
print(y_train.value_counts())

print("\nTesting labels:")
print(y_test.value_counts())


# ============================================================
# 4. CREATE MODEL
# ============================================================

model = XGBClassifier(
    n_estimators=200,
    max_depth=5,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42,
    eval_metric="logloss"
)


# ============================================================
# 5. TRAIN MODEL
# ============================================================

print("\n========================================")
print("TRAINING MODEL")
print("========================================")

model.fit(
    X_train,
    y_train
)

print("Training completed.")


# ============================================================
# 6. PREDICTIONS
# ============================================================

y_pred = model.predict(X_test)

y_probability = model.predict_proba(X_test)[:, 1]


# ============================================================
# 7. EVALUATION
# ============================================================

print("\n========================================")
print("MODEL PERFORMANCE")
print("========================================")

print(
    "Accuracy:",
    round(accuracy_score(y_test, y_pred), 4)
)

print(
    "Precision:",
    round(precision_score(y_test, y_pred), 4)
)

print(
    "Recall:",
    round(recall_score(y_test, y_pred), 4)
)

print(
    "F1:",
    round(f1_score(y_test, y_pred), 4)
)


print("\nClassification Report:")
print(
    classification_report(
        y_test,
        y_pred,
        target_names=[
            "Non-Match",
            "Match"
        ]
    )
)


# ============================================================
# 8. CONFUSION MATRIX
# ============================================================

print("\nConfusion Matrix:")

print(
    confusion_matrix(
        y_test,
        y_pred
    )
)


# ============================================================
# 9. FEATURE IMPORTANCE
# ============================================================

importance = pd.DataFrame({
    "feature": feature_columns,
    "importance": model.feature_importances_
})

importance = importance.sort_values(
    "importance",
    ascending=False
)

print("\n========================================")
print("FEATURE IMPORTANCE")
print("========================================")

print(
    importance.to_string(index=False)
)


# ============================================================
# 10. SAVE MODEL
# ============================================================

MODEL_FILE = "entity_resolution_xgboost_model.pkl"

joblib.dump(
    model,
    MODEL_FILE
)

print("\nModel saved as:", MODEL_FILE)


# ============================================================
# 11. CREATE TEST RESULTS
# ============================================================

test_results = df.loc[
    X_test.index,
    [
        "id_1",
        "id_2"
    ]
].copy()

test_results["actual_label"] = y_test

test_results["predicted_label"] = y_pred

test_results["match_probability"] = y_probability


test_results.to_csv(
    "entity_resolution_test_results.csv",
    index=False
)

print(
    "Test results saved as:",
    "entity_resolution_test_results.csv"
)
