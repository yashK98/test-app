import json
import logging

from app.src.models.ValidationResponse import ValidationResponse
from app.src.prompts.ValidationPrompt import ValidationPrompt
from app.src.service.DocumentService import DocumentService
from app.src.service.LLMService import LLMService
from app.src.service.PolicyService import PolicyService

logger = logging.getLogger(__name__)


class AgentService:

    def __init__(self):

        self.document_service = DocumentService()
        self.policy_service = PolicyService()
        self.llm_service = LLMService()

    def validate_document(self, document_path: str) -> ValidationResponse:
        """
        Validates a document against the configured policy.
        """

        logger.info(f"Validating {document_path}")

        document = self.document_service.extract_text(document_path)

        policy = self.policy_service.load_policy()

        prompt = ValidationPrompt.build(
            policy=policy,
            document=document
        )

        response = self.llm_service.invoke(prompt)

        logger.info("Received response from LLM")

        try:

            response_json = json.loads(response)

            return ValidationResponse(
                approved=response_json["approved"],
                confidence=response_json["confidence"],
                reason=response_json["reason"],
                violations=response_json["violations"]
            )

        except Exception as ex:

            logger.exception(ex)

            raise Exception(
                f"Unable to parse LLM response.\n\n{response}"
            )
