import pandas as pd
from sqlalchemy import create_engine, text

ccs_engine = create_engine(CCS_CONNECTION_STRING)
aicoe_engine = create_engine(AICOE_CONNECTION_STRING)

# 1. Fetch case IDs from AICOE
case_ids = pd.read_sql(
    "SELECT case_id FROM your_aicoe_table",
    aicoe_engine
)

# 2. Fetch corresponding values from CCS in batches
batch_size = 5000
results = []

with ccs_engine.connect() as conn:
    for i in range(0, len(case_ids), batch_size):

        batch = case_ids["case_id"].iloc[i:i + batch_size].tolist()

        query = text("""
            SELECT case_id, isquickkill
            FROM your_ccs_table
            WHERE case_id IN :case_ids
        """).bindparams(
            __import__('sqlalchemy').bindparam(
                'case_ids', expanding=True
            )
        )

        df = pd.read_sql(query, conn, params={"case_ids": batch})
        results.append(df)

# 3. Combine results
updated_data = pd.concat(results, ignore_index=True)

# 4. Bulk update AICOE
update_query = text("""
    UPDATE your_aicoe_table
    SET isquickkill = :isquickkill
    WHERE case_id = :case_id
""")

records = updated_data.to_dict(orient="records")

with aicoe_engine.begin() as conn:
    conn.execute(update_query, records)

print(f"Updated {len(records)} records")
