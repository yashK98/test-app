
import re


# ============================================================
# EMAIL THREAD BOUNDARY PATTERNS
# ============================================================

BOUNDARY_PATTERN = re.compile(
    r"(?im)"
    r"^[ \t]*[-_ ]*Original Message[-_ ]*[ \t]*$"
    r"|^[ \t]*Begin forwarded message:[ \t]*$"
    r"|^[ \t]*-{2,}[ \t]*Forwarded message[ \t]*-{2,}[ \t]*$"
    r"|^[ \t]*On .{5,200}wrote:[ \t]*$"
    r"|^[ \t]*[-_]{5,}[ \t]*$"
)


# ============================================================
# EMAIL HEADER PATTERNS
# ============================================================

HEADER_BLOCK_PATTERN = re.compile(
    r"(?im)^[ \t]*From:[ \t]?.{0,1500}?"
    r"^[ \t]*Subject:[ \t]?.*(?:\n|$)"
)


HEADER_PATTERN = re.compile(
    r"(?im)^[ \t]*(?:From|Sent|Date|To|Cc|Bcc|Subject):.*$"
)


# ============================================================
# EMAIL SIGNATURE PATTERN
# ============================================================

SIGNATURE_RE = re.compile(
    r"(?i)^[ \t]*"
    r"(?:(?:warm|best|kind|many)\s+)?"
    r"(?:regards|thanks|thank you|"
    r"thanks and regards|thanks & regards|"
    r"sincerely|cheers)"
    r"[,.!?]?[ \t]*$"
)


# ============================================================
# PII PATTERNS
# ============================================================

PII_PATTERNS = [

    # Email addresses
    (
        re.compile(
            r"\b[\w.+-]+@[\w-]+(?:\.[\w-]+)+\b"
        ),
        "[EMAIL]"
    ),

    # Credit / debit card numbers
    (
        re.compile(
            r"\b(?:\d[ -]?){13,19}\b"
        ),
        "[CARD_NUMBER]"
    ),

    # Phone numbers
    (
        re.compile(
            r"(?<!\w)\+?\d[\d\s().-]{8,15}\d(?!\w)"
        ),
        "[PHONE]"
    ),

    # IBAN
    (
        re.compile(
            r"\b[A-Z]{2}\d{2}[A-Z0-9]{11,30}\b"
        ),
        "[IBAN]"
    ),

    # PAN
    (
        re.compile(
            r"\b[A-Z]{5}\d{4}[A-Z]\b"
        ),
        "[PAN]"
    ),

    # Names in common email greetings
    (
        re.compile(
            r"(?im)^(Hi|Hello|Dear|Hey)\s+"
            r"[A-Z][\w.'-]*"
            r"(?:\s+[A-Z][\w.'-]*){0,2}"
            r"(?=\s*[,\n])"
        ),
        r"\1 [NAME]"
    )
]


# ============================================================
# CONTEXTUAL ACCOUNT NUMBER DETECTION
# ============================================================

ACCOUNT_PATTERN = re.compile(
    r"(?i)(\b(?:account|acct|a/c)"
    r"(?:\s*(?:number|no\.?|#))?\s*[:#-]?\s*)"
    r"([A-Z0-9-]{8,20})"
)
