"""
Examples demonstrating usage of Pathfinder Platform SDK MinIOClient.
"""

from pathfinder_platform_sdk.minio import MinIOClient

###########################################################################
# Create Client
###########################################################################

client = MinIOClient(
    endpoint="https://minio.example.com",
    access_key="ACCESS_KEY",
    secret_key="SECRET_KEY",
    verify=False,
    bucket_name="documents"
)

###########################################################################
# Example 1 - Upload a Local File
###########################################################################

client.upload_minio_file(
    source_path="/tmp",
    filename="invoice.pdf",
    object_name="documents/invoice.pdf"
)

###########################################################################
# Example 2 - Download a File
###########################################################################

client.download_minio_file(
    object_name="documents/invoice.pdf",
    download_path="/tmp/invoice.pdf"
)

###########################################################################
# Example 3 - Upload Generated Bytes
###########################################################################

with open("report.pdf", "rb") as f:
    file_bytes = f.read()

client.upload_filedata(
    file_data=file_bytes,
    object_name="reports/report.pdf"
)

###########################################################################
# Example 4 - Read Object Contents
###########################################################################

content = client.read_minio_file(
    object_name="reports/report.pdf"
)

print(f"Read {len(content)} bytes")

###########################################################################
# Example 5 - List Objects
###########################################################################

objects = client.list_all_objects()

for obj in objects:
    print(obj)

###########################################################################
# Example 6 - Delete Object
###########################################################################

client.remove_object(
    object_name="reports/report.pdf"
)

###########################################################################
# Example 7 - Upload Generated CSV
###########################################################################

csv_data = b"""id,name
1,Alice
2,Bob
"""

client.upload_filedata(
    file_data=csv_data,
    object_name="exports/users.csv"
)

###########################################################################
# Example 8 - Read JSON File
###########################################################################

json_bytes = client.read_minio_file(
    object_name="config/application.json"
)

print(json_bytes.decode("utf-8"))

###########################################################################
# Example 9 - Archive Log File
###########################################################################

client.upload_minio_file(
    source_path="/var/log",
    filename="application.log",
    object_name="archive/application.log"
)
