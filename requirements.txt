SECURITY_WARNING_PATTERNS = [
    # External email warnings
    r"(?im)^\s*ATTENTION:\s*This email came from an external source.*?(?=\nDear|\nHi|\nHello|\Z)",

    # Phishing warnings
    r"(?im)^\s*Always report suspicious emails.*$",

    # Attachment warnings
    r"(?im)^\s*Do not open attachments from unknown senders.*$",

    # Confidentiality notices
    r"(?im)^\s*This (?:email|message) (?:is|may be) confidential.*$",

    # Accidental recipient notices
    r"(?im)^\s*If you (?:have )?received this (?:email|message) in error.*$",

    # Email encryption notices
    r"(?im)^\s*This message is protected by Microsoft.*$",
]


r"(?is)\bATTENTION:\s*This email came from an external source\b.*?to protect the Bank and our clients\.?"
