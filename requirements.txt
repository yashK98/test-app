all_predictions = {}
all_actual = {}

print("\n" + "=" * 60)

for i, attribute in enumerate(attributes):

    # Get predictions for current classification head
    predictions = np.argmax(logits[i], axis=-1)

    # Get actual labels for current attribute
    actual = actual_labels[i]

    all_predictions[attribute] = predictions
    all_actual[attribute] = actual

    # Calculate metrics
    print("\n" + "-" * 60)
    print(f"ATTRIBUTE: {attribute}")
    print("-" * 60)

    print(
        "Accuracy:",
        accuracy_score(actual, predictions)
    )

    print("\nClassification Report:")

    print(
        classification_report(
            actual,
            predictions,
            zero_division=0
        )
    )

    print("\nConfusion Matrix:")

    print(
        confusion_matrix(actual, predictions)
    )


# Overall exact match accuracy
actual_matrix = np.column_stack([
    all_actual[attr] for attr in attributes
])

prediction_matrix = np.column_stack([
    all_predictions[attr] for attr in attributes
])

exact_match = np.all(
    actual_matrix == prediction_matrix,
    axis=1
